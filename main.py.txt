def db_connect() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_FILE, timeout=30)
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA busy_timeout=30000")
    conn.row_factory = sqlite3.Row
    return conn

def now_iso() -> str:
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")


def current_month() -> str:
    return datetime.now().strftime("%Y-%m")


# ─── RATE LIMITER ──────────────────────────────────────────────

def rate_limit_check(user_id: int) -> bool:
    """Возвращает True если запрос разрешён, False если превышен лимит."""
    conn = db_connect()
    c = conn.cursor()
    now = time.time()
    window = now - 60  # последняя минута

    c.execute("DELETE FROM rate_limit WHERE window_start < ?", (window,))
    c.execute("SELECT COUNT(*) FROM rate_limit WHERE user_id = ?", (user_id,))
    count = c.fetchone()[0]

    if count >= RATE_LIMIT_PER_MIN:
        conn.commit()
        conn.close()
        return False

    c.execute("INSERT INTO rate_limit (user_id, window_start) VALUES (?, ?)", (user_id, now))
    conn.commit()
    conn.close()
    return True


# ─── USER HELPERS ──────────────────────────────────────────────

def get_user(user_id: int) -> Optional[dict]:
    conn = db_connect()
    c = conn.cursor()
    c.execute("SELECT * FROM users WHERE user_id = ?", (user_id,))
    row = c.fetchone()
    conn.close()
    return dict(row) if row else None


def is_deleted(user_id: int) -> bool:
    conn = db_connect()
    c = conn.cursor()
    c.execute("SELECT 1 FROM deleted_users WHERE user_id = ?", (user_id,))
    row = c.fetchone()
    conn.close()
    return row is not None


def create_user(user_id: int):
    conn = db_connect()
    c = conn.cursor()
    c.execute("""
        INSERT OR IGNORE INTO users (user_id, created_at, last_active, ref_bonus_month)
        VALUES (?, ?, ?, ?)
    """, (user_id, now_iso(), now_iso(), current_month()))
    conn.commit()
    conn.close()


def update_user(user_id: int, **kwargs):
    conn = db_connect()
    c = conn.cursor()
    sets = ", ".join(f"{k} = ?" for k in kwargs)
    vals = list(kwargs.values()) + [user_id]
    c.execute(f"UPDATE users SET {sets} WHERE user_id = ?", vals)
    conn.commit()
    conn.close()


def check_sub_active(user: dict) -> bool:
    """Возвращает True если триал или подписка активны."""
    now = datetime.now()
    trial_end = user.get("trial_end")
    sub_end = user.get("sub_end")
    if trial_end:
        try:
            t = datetime.strptime(trial_end, "%Y-%m-%d %H:%M:%S")
            if now < t:
                return True
        except (ValueError, TypeError):
            pass
    if sub_end:
        try:
            s = datetime.strptime(sub_end, "%Y-%m-%d %H:%M:%S")
            if now < s:
                return True
        except (ValueError, TypeError):
            pass
    return False

# ─── STEAM MARKET API ──────────────────────────────────────────

_market_cache = {}  # key -> (timestamp, data)


async def fetch_steam_market(appid: int, count: int = 50) -> list:
    """Получает предметы с Steam Market через search/render endpoint."""
    cache_key = f"market_{appid}"
    now = time.time()
    if cache_key in _market_cache:
        ts, data = _market_cache[cache_key]
        if now - ts < CACHE_TTL:
            return data

    url = "https://steamcommunity.com/market/search/render/"
    params = {
        "query": "",
        "start": 0,
        "count": count,
        "search_descriptions": 0,
        "sort_column": "price",
        "sort_dir": "asc",
        "appid": appid,
        "norender": 1,
        "currency": 5,
        "country": "RU",
    }
    headers = {
        "User-Agent": "Mozilla/5.0",
        "Accept-Language": "ru-RU,ru;q=0.9",
    }

    try:
        async with aiohttp.ClientSession() as session:
            async with session.get(url, params=params, headers=headers, timeout=10) as resp:
                if resp.status == 429:
                    logger.warning("Steam rate limit (429), backing off")
                    return []
                if resp.status != 200:
                    logger.warning(f"Steam returned {resp.status}")
                    return []
                data = await resp.json(content_type=None)

        results = []
        for item in data.get("results", []):
            sell_price = item.get("sell_price", 0)
            if sell_price == 0:
                continue
            name = item.get("hash_name", item.get("name", "Unknown"))
            listings = item.get("sell_listings", 0)
            asset = item.get("asset_description", {})
            icon_url = asset.get("icon_url", "")
            icon_url_large = asset.get("icon_url_large", "")
            color = asset.get("name_color", "FFFFFF")

            results.append({
                "name": name,
                "price": sell_price / 100,
                "listings": listings,
                "icon_url": icon_url,
                "icon_url_large": icon_url_large,
                "color": color,
                "appid": appid,
            })

        _market_cache[cache_key] = (now, results)
        return results

    except asyncio.TimeoutError:
        logger.warning("Steam API timeout")
        return []
    except Exception as e:
        logger.warning(f"Steam API error: {e}")
        return []


async def fetch_steam_priceoverview(appid: int, market_hash_name: str) -> Optional[dict]:
    """Получает текущую цену конкретного предмета."""
    url = "https://steamcommunity.com/market/priceoverview/"
    params = {
        "appid": appid,
        "currency": 5,
        "market_hash_name": market_hash_name,
    }
    headers = {"User-Agent": "Mozilla/5.0"}

    try:
        async with aiohttp.ClientSession() as session:
            async with session.get(url, params=params, headers=headers, timeout=10) as resp:
                if resp.status != 200:
                    return None
                data = await resp.json(content_type=None)

        lowest = data.get("lowest_price", "")
        median = data.get("median_price", "")
        volume = data.get("volume", "0")

        price = None
        if lowest:
            clean = re.sub(r"[^\d.]", "", lowest.replace(",", "."))
            if clean:
                price = float(clean)

        return {"price": price, "volume": volume, "lowest_text": lowest}

    except Exception:
        return None

# ─── STEAM ID VALIDATION ──────────────────────────────────────

def validate_steam_id(text: str) -> Optional[str]:
    text = text.strip()
    # Прямой SteamID64
    if re.match(r"^\d{17}$", text):
        return text
    # Ссылка на профиль
    m = re.search(r"steamcommunity\.com/profiles/(\d{17})", text)
    if m:
        return m.group(1)
    # Короткая ссылка
    if "steamcommunity.com/id/" in text:
        return text  # сохраняем как есть, позже можно резолвить
    return None

async def cmd_start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id

    if not rate_limit_check(user_id):
        await update.message.reply_text("⏳ Слишком много запросов. Подождите минуту.")
        return

    create_user(user_id)
    user = get_user(user_id)

    # Проверяем удалённых
    if is_deleted(user_id):
        await update.message.reply_text(
            "Вы удалили свои данные. Обратитесь к администратору для восстановления."
        )
        return

    # Рефералка
    ref_id = None
    if context.args:
        arg = context.args[0]
        if arg.startswith("ref"):
            try:
                ref_id = int(arg[3:])
            except ValueError:
                pass

    # Защита от самореферала
    if ref_id == user_id:
        ref_id = None

    # Проверяем, что реферер существует
    if ref_id:
        ref_user = get_user(ref_id)
        if not ref_user:
            ref_id = None

    # Если юзер уже прошёл настройку
    if user and user.get("setup_done"):
        update_user(user_id, last_active=now_iso())
        await show_main_menu(update, context, user_id)
        return

    # Сохраняем реферера (только для новых юзеров)
    if ref_id and not user.get("referrer_id"):
        update_user(user_id, referrer_id=ref_id)
        # Уведомляем реферера
        try:
            await context.bot.send_message(
                ref_id,
                "👋 По вашей ссылке кто-то зашёл в бот!\n"
                "Когда он завершит настройку и сделает первую покупку — вы получите +1 день."
            )
        except Exception:
            pass
        # Показываем новичку
        await update.message.reply_text(
            "🎉 Вас пригласил друг!\n"
            "После настройки вы получите +1 день к пробной версии.\n\n"
        )
    elif ref_id and user.get("referrer_id") == ref_id:
        pass  # уже привязан

    # Политика конфиденциальности
    if not user.get("agreed"):
        text = (
            "👋 Добро пожаловать в EarnSkin!\n\n"
            "Перед началом ознакомьтесь с политикой конфиденциальности:\n\n"
            "📌 Мы храним: ваш Telegram ID, Steam ID, выбранные настройки.\n"
            "📌 Мы НЕ храним: пароли, платёжные данные, личные переписки.\n"
            "📌 Данные используются только для подбора предметов и работы бота.\n"
            "📌 Вы можете удалить все данные командой /delete.\n\n"
            "Согласны продолжить?"
        )
        await update.message.reply_text(text, reply_markup=kb_consent())
        return

    # Если согласился, но не прошёл туториал
    if user.get("agreed") and not user.get("tutorial_step"):
        await show_tutorial(update, context, 1)
        return

    # Если прошёл туториал, но не привязал Steam
    if user.get("agreed") and user.get("tutorial_step") and not user.get("steam_id"):
        await update.message.reply_text(
            "🔗 Введите ваш Steam ID или ссылку на профиль Steam:\n"
            "Например: 76561198000000000 или https://steamcommunity.com/profiles/76561198000000000"
        )
        return

    await show_main_menu(update, context, user_id)

# ─── HANDLERS: CALLBACKS ──────────────────────────────────────

async def callback_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    user_id = query.from_user.id
    data = query.data

    if not rate_limit_check(user_id):
        await query.message.reply_text("⏳ Слишком много запросов. Подождите минуту.")
        return

    user = get_user(user_id)
    if not user:
        create_user(user_id)
        user = get_user(user_id)

    # ─── Consent ───
    if data == "consent_yes":
        update_user(user_id, agreed=1)
        await show_tutorial(update, context, 1)
        return

    if data == "consent_no":
        await query.message.edit_text(
            "Без согласия с политикой бот работать не может.\n"
            "Вы можете удалить данные: /delete"
        )
        return

    # ─── Tutorial ───
    if data == "tut_skip":
        update_user(user_id, tutorial_step=3, agreed=1)
        await query.message.edit_text(
            "🔗 Введите ваш Steam ID или ссылку на профиль Steam:\n"
            "Например: 76561198000000000 или https://steamcommunity.com/profiles/76561198000000000"
        )
        return

    if data.startswith("tut_"):
        step = int(data.split("_")[1])
        if step <= 3:
            await show_tutorial(update, context, step)
        elif data == "tut_done":
            update_user(user_id, tutorial_step=3, agreed=1)
            await query.message.edit_text(
                "🔗 Введите ваш Steam ID или ссылку на профиль Steam:\n"
                "Например: 76561198000000000 или https://steamcommunity.com/profiles/76561198000000000"
            )
        return

    # ─── Game selection ───
    if data.startswith("game_"):
        game_key = data.split("_")[1]
        if game_key in GAMES:
            update_user(user_id, game=game_key)
            if not user.get("setup_done"):
                await query.message.edit_text(
                    f"🎮 Игра: {GAMES[game_key]['name']}\n\n"
                    "Выберите ценовой сегмент:",
                    reply_markup=kb_segments()

                    f"✅ Игра изменена на {GAMES[game_key]['name']}",

    if data.startswith("seg_"):
        seg_key = data
        if seg_key in SEGMENTS:
            update_user(user_id, price_segment=seg_key)

            if not user.get("setup_done"):
                # Завершение настройки
                extend_trial(user_id, TRIAL_HOURS)

                # Реферальный бонус новичку
                if user.get("referrer_id"):

                                ref_id,

                    f"✅ Сегмент изменён: {SEGMENTS[seg_key]['label']}",

# ─── FIND ITEMS ───────────────────────────────────────────────

async def handle_find_items(update: Update, context: ContextTypes.DEFAULT_TYPE, user_id: int, offset: int = 0):
    user = get_user(user_id)
    if not user or not user.get("setup_done"):
        await update.callback_query.message.edit_text("Сначала завершите настройку: /start")
        return

    if not check_sub_active(user):
        await update.callback_query.message.edit_text(
            "❌ Подписка истекла. Оформите в разделе «💳 Подписка».",
            reply_markup=kb_subscription()
        )
        return

    game_key = user.get("game", "cs2")
    seg_key = user.get("price_segment", "seg_1")
    game = GAMES.get(game_key, GAMES["cs2"])

    items = await fetch_steam_market(game["appid"], count=100)
    items = filter_by_segment(items, seg_key)
    items = sorted(items, key=lambda x: x["price"])

    if not items:
        await update.callback_query.message.edit_text(
            "🔍 Предметы не найдены. Попробуйте другой сегмент в настройках.",
            reply_markup=kb_back()
        )
        return

    context.user_data["last_items"] = items
    context.user_data["item_offset"] = offset

    page = items[offset:offset+5]
    text_parts = ["🔍 Найденные предметы:\n"]
    for i, item in enumerate(page):
        idx = offset + i
        # Считаем потенциальный ROI (целевая +15%)
        target = item["price"] * 1.15
        roi = 15.0
        text_parts.append(
            f"{idx+1}. {item['name']}\n"
            f"   💵 {item['price']:.2f} ₽ | 📦 {item['listings']} лист.\n"
            f"   📈 Потенциал: +{roi:.0f}% → {target:.2f} ₽\n"
        )

    buttons = []
    for i, item in enumerate(page):
        idx = offset + i
        buttons.append([
            InlineKeyboardButton(f"⭐ {idx+1}", callback_data=f"fav_{idx}"),
            InlineKeyboardButton(f"💰 {idx+1}", callback_data=f"buy_{idx}"),
        ])

    nav = []
    if offset > 0:
        nav.append(InlineKeyboardButton("⬅️ Назад", callback_data=f"page_{max(0, offset-5)}"))
    if offset + 5 < len(items):
        nav.append(InlineKeyboardButton("➡️ Ещё", callback_data="more_items"))
    nav.append(InlineKeyboardButton("🏠 Меню", callback_data="back_menu"))
    buttons.append(nav)

    await update.callback_query.message.edit_text(
        "\n".join(text_parts),
        reply_markup=InlineKeyboardMarkup(buttons)
    )

# ─── STATS ────────────────────────────────────────────────────

async def handle_stats(update: Update, context: ContextTypes.DEFAULT_TYPE, user_id: int):
    user = get_user(user_id)
    conn = db_connect()
    c = conn.cursor()

    bought = c.execute("SELECT COUNT(*) FROM bought_items WHERE user_id=?", (user_id,)).fetchone()[0]
    sold = c.execute("SELECT COUNT(*) FROM bought_items WHERE user_id=? AND status='sold'", (user_id,)).fetchone()[0]
    active = c.execute("SELECT COUNT(*) FROM bought_items WHERE user_id=? AND status='active'", (user_id,)).fetchone()[0]
    favs = c.execute("SELECT COUNT(*) FROM favorites WHERE user_id=?", (user_id,)).fetchone()[0]

    # Считаем реальную прибыль
    sold_rows = c.execute(
        "SELECT buy_price, sold_price FROM bought_items WHERE user_id=? AND status='sold'",
        (user_id,)
    ).fetchall()
    total_profit = 0
    total_roi = 0
    for row in sold_rows:
        r = dict(row)
        bp = r["buy_price"] or 0
        sp = r["sold_price"] or 0
        profit = sp - bp
        total_profit += profit
        if bp > 0:
            total_roi += (profit / bp) * 100

    avg_roi = total_roi / len(sold_rows) if sold_rows else 0

    # Всего юзеров
    total_users = c.execute("SELECT COUNT(*) FROM users WHERE setup_done=1").fetchone()[0]

    conn.close()

    text = (
        "📊 Статистика\n\n"
        f"💼 Всего покупок: {bought}\n"
        f"✅ Продано: {sold}\n"
        f"⏳ Активных: {active}\n"
        f"⭐ В избранном: {favs}\n"
        f"💰 Общая прибыль: {total_profit:+.2f} ₽\n"
        f"📈 Средний ROI: {avg_roi:.1f}%\n\n"
        f"👥 Всего пользователей в боте: {total_users}"
    )
    await update.callback_query.message.edit_text(text, reply_markup=kb_back())

# ─── BUY ITEM ─────────────────────────────────────────────────

async def _buy_item(update: Update, context: ContextTypes.DEFAULT_TYPE, user_id: int, name: str, price: float):
    conn = db_connect()
    c = conn.cursor()

    count = c.execute("SELECT COUNT(*) FROM bought_items WHERE user_id=? AND status='active'", (user_id,)).fetchone()[0]
    if count >= 20:
        await query_message(update, "💼 Портфель заполнен (максимум 20). Сначала продайте что-нибудь.")
        conn.close()
        return

    sell_target = price * 1.15
    c.execute(
        "INSERT INTO bought_items (user_id, item_name, buy_price, sell_target, status, bought_at) "
        "VALUES (?,?,?,?, 'active', ?)",
        (user_id, name, price, sell_target, now_iso())
    )
    conn.commit()
    conn.close()

    # Реферальный бонус после первой покупки
    user = get_user(user_id)
    if user and user.get("referrer_id"):
        ref_id = user["referrer_id"]
        ref_user = get_user(ref_id)
        if ref_user:
            ref_month = ref_user.get("ref_bonus_month") or current_month()
            ref_count = ref_user.get("ref_bonus_count", 0)
            if ref_month != current_month():
                ref_count = 0
                ref_month = current_month()
            if ref_count < REF_MAX_BONUS_PER_MONTH:
                extend_trial(ref_id, REF_BONUS_HOURS)
                update_user(ref_id, ref_bonus_count=ref_count + 1, ref_bonus_month=ref_month)
                try:
                    await context.bot.send_message(
                        ref_id,
                        f"🎉 Ваш реферал совершил первую покупку!\n"
                        f"Вам начислено +1 день.\n"
                        f"Рефералов в этом месяце: {ref_count + 1}/{REF_MAX_BONUS_PER_MONTH}"
                    )
                except Exception:
                    pass

    await query_message(update, (
        f"✅ Предмет добавлен в портфель!\n\n"
        f"📦 {name}\n"
        f"💵 Цена покупки: {price:.2f} ₽\n"
        f"🎯 Целевая цена: {sell_target:.2f} ₽ (+15%)\n\n"
        f"💡 Бот будет следить за ценой и уведомит, когда пора продавать."
    ))


async def query_message(update: Update, text: str):
    """Отправляет сообщение в ответ на callback или message."""
    if hasattr(update, "callback_query") and update.callback_query:
        await update.callback_query.message.reply_text(text, reply_markup=kb_back())
    else:
        await update.message.reply_text(text, reply_markup=kb_back())


# ─── TEXT HANDLER (Steam ID, Promo) ───────────────────────────

async def text_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    text = update.message.text.strip()

    if not rate_limit_check(user_id):
        await update.message.reply_text("⏳ Слишком много запросов. Подождите минуту.")
        return

    # Steam ID ввод
    if context.user_data.get("awaiting_steam") or not get_user(user_id).get("steam_id"):
        steam_id = validate_steam_id(text)
        if not steam_id:
            await update.message.reply_text(
                "❌ Неверный формат. Введите Steam ID (17 цифр) или ссылку на профиль."
            )
            return

        # Проверка дублей Steam ID
        conn = db_connect()
        c = conn.cursor()
        existing = c.execute("SELECT 1 FROM users WHERE steam_id=? AND user_id!=?", (steam_id, user_id)).fetchone()
        if existing:
            conn.close()
            await update.message.reply_text(
                "❌ Этот Steam ID уже привязан к другому аккаунту."
            )
            return
        conn.close()

        context.user_data["awaiting_steam"] = False
        update_user(user_id, steam_id=steam_id)

        user = get_user(user_id)
        if not user.get("game"):
            await update.message.reply_text(
                f"✅ Steam ID сохранён!\n\nВыберите игру:",
                reply_markup=kb_games()
            )
        else:
            await update.message.reply_text("✅ Steam ID обновлён!", reply_markup=kb_back())
        return

    # Промокод ввод
    if context.user_data.get("awaiting_promo"):
        context.user_data["awaiting_promo"] = False
        await _redeem_promo(update, context, user_id, text)
        return

    # Fallback
    await update.message.reply_text(
        "Я не понимаю это сообщение. Используйте кнопки меню или команды.\n"
        "/menu — главное меню\n/help — справка"
    )


async def _redeem_promo(update: Update, context: ContextTypes.DEFAULT_TYPE, user_id: int, code: str):
    conn = db_connect()
    c = conn.cursor()
    row = c.execute("SELECT * FROM promocodes WHERE code=?", (code,)).fetchone()
    if not row:
        conn.close()
        await update.message.reply_text("❌ Промокод не найден.", reply_markup=kb_back())
        return

    promo = dict(row)
    if promo["uses"] >= promo["max_uses"]:
        conn.close()
        await update.message.reply_text("❌ Промокод исчерпан.", reply_markup=kb_back())
        return

    if promo.get("expires_at"):
        try:
            exp = datetime.strptime(promo["expires_at"], "%Y-%m-%d %H:%M:%S")
            if datetime.now() > exp:
                conn.close()
                await update.message.reply_text("❌ Промокод истёк.", reply_markup=kb_back())
                return
        except (ValueError, TypeError):
            pass

    already = c.execute("SELECT 1 FROM promo_redemptions WHERE user_id=? AND code=?", (user_id, code)).fetchone()
    if already:
        conn.close()
        await update.message.reply_text("❌ Вы уже активировали этот промокод.", reply_markup=kb_back())
        return

    c.execute("UPDATE promocodes SET uses = uses + 1 WHERE code=?", (code,))
    c.execute("INSERT INTO promo_redemptions (user_id, code, redeemed_at) VALUES (?,?,?)",
              (user_id, code, now_iso()))
    conn.commit()
    conn.close()

    extend_sub(user_id, promo["days"])
    await update.message.reply_text(
        f"✅ Промокод активирован!\nПодписка продлена на {promo['days']} дней.",
        reply_markup=kb_back()
    )

# ─── BACKGROUND JOBS ──────────────────────────────────────────

async def job_new_items_check(context: ContextTypes.DEFAULT_TYPE):
    """Каждые 2 минуты — проверка новых предметов для юзеров с включёнными уведомлениями."""
    conn = db_connect()
    c = conn.cursor()
    rows = c.execute(
        "SELECT user_id, game, price_segment FROM users WHERE setup_done=1 AND notifications=1"
    ).fetchall()
    conn.close()

    for row in rows:
        r = dict(row)
        user_id = r["user_id"]
        user = get_user(user_id)
        if not user or not check_sub_active(user):
            continue

        game_key = r.get("game", "cs2")
        seg_key = r.get("price_segment", "seg_1")
        game = GAMES.get(game_key, GAMES["cs2"])

        items = await fetch_steam_market(game["appid"], count=30)
        items = filter_by_segment(items, seg_key)
        if not items:
            continue

        # Берём первый предмет, не отправленный за 24 часа
        for item in items:
            item_name = item["name"]
            conn = db_connect()
            c = conn.cursor()
            recent = c.execute(
                "SELECT 1 FROM notified_items WHERE user_id=? AND item_name=? AND "
                "notified_at > datetime('now','-1 day')",
                (user_id, item_name)
            ).fetchone()
            if not recent:
                # Проверяем, не в избранном и не в портфеле
                in_fav = c.execute("SELECT 1 FROM favorites WHERE user_id=? AND item_name=?", (user_id, item_name)).fetchone()
                in_bought = c.execute("SELECT 1 FROM bought_items WHERE user_id=? AND item_name=? AND status='active'", (user_id, item_name)).fetchone()
                if in_fav or in_bought:
                    conn.close()
                    continue
                c.execute("INSERT INTO notified_items (user_id, item_name, notified_at) VALUES (?,?,?)",
                          (user_id, item_name, now_iso()))
                conn.commit()
                conn.close()

                target = item["price"] * 1.15
                text = (
                    f"🔔 Новый выгодный предмет!\n\n"
                    f"📦 {item_name}\n"
                    f"💵 Цена: {item['price']:.2f} ₽\n"
                    f"📦 Листингов: {item['listings']}\n"
                    f"📈 Потенциал: +15% → {target:.2f} ₽\n\n"
                    f"⚡ Быстрее за покупкой — пока предмет доступен!"
                )
                try:
                    await context.bot.send_message(user_id, text)
                except Exception:
                    pass
                break
            conn.close()

        await asyncio.sleep(0.5)


async def job_auto_sell_check(context: ContextTypes.DEFAULT_TYPE):
    """Каждый час — проверка цен купленных предметов."""
    conn = db_connect()
    c = conn.cursor()
    rows = c.execute(
        "SELECT id, user_id, item_name, buy_price, sell_target, appid "
        "FROM bought_items WHERE status='active'"
    ).fetchall()
    conn.close()

    for row in rows:
        r = dict(row)
        user_id = r["user_id"]
        user = get_user(user_id)
        if not user or not check_sub_active(user):
            continue

        game_key = user.get("game", "cs2")
        game = GAMES.get(game_key, GAMES["cs2"])

        price_data = await fetch_steam_priceoverview(game["appid"], r["item_name"])
        if not price_data or price_data.get("price") is None:
            continue

        current_price = price_data["price"]
        if current_price >= r["sell_target"]:
            profit = current_price - r["buy_price"]
            roi_pct = (profit / r["buy_price"]) * 100 if r["buy_price"] > 0 else 0

            conn = db_connect()
            c = conn.cursor()
            c.execute(
                "UPDATE bought_items SET status='sold', sold_at=?, sold_price=? WHERE id=?",
                (now_iso(), current_price, r["id"])
            )
            conn.commit()
            conn.close()

            try:
                await context.bot.send_message(
                    user_id,
                    f"✅ Время продавать!\n\n"
                    f"📦 {r['item_name']}\n"
                    f"💵 Куплено: {r['buy_price']:.2f} ₽\n"
                    f"💵 Текущая цена: {current_price:.2f} ₽\n"
                    f"💰 Прибыль: {profit:+.2f} ₽\n"
                    f"📈 ROI: {roi_pct:+.1f}%\n\n"
                    f"🎯 Целевая цена достигнута — фиксируйте прибыль!"
                )
            except Exception:
                pass

        await asyncio.sleep(1)


# ─── GRACEFUL SHUTDOWN ────────────────────────────────────────

def setup_signal_handlers(app):
    def handler(signum, frame):
        logger.info(f"Signal {signum} received, shutting down...")
        try:
            conn = db_connect()
            conn.execute("PRAGMA wal_checkpoint(TRUNCATE)")
            conn.close()
        except Exception:
            pass

    signal.signal(signal.SIGTERM, handler)
    signal.signal(signal.SIGINT, handler)

    app.add_handler(PreCheckoutQueryHandler(precheckout_handler))
    app.add_handler(MessageHandler(filters.SUCCESSFUL_PAYMENT, successful_payment_handler))

    # Signals
    setup_signal_handlers(app)

    logger.info("EarnSkin bot started!")
    app.run_polling(allowed_updates=Update.ALL_TYPES)
