async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    await update.message.reply_html(
        f"Привет, {user.mention_html()}! 👋\n\n"
        "Это EarnSkin — бот для анализа и торговли скинами.\n"
        "Нажми /tutorial, чтобы пройти быстрый туториал."
    )

async def tutorial(update: Update, context: ContextTypes.DEFAULT_TYPE):

async def admin(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if update.effective_user.id != ADMIN_ID:
        await update.message.reply_text("❌ Доступ запрещён. Это админ-команда.")
        return
    await update.message.reply_text(
        "🛠 Панель администратора EarnSkin\n\n"
        "Здесь будут: статистика, логи, список пользователей, бэкапы."
    )
