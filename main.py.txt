async def get_user(user_id: int):
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        cur = await db.execute("SELECT * FROM users WHERE user_id = ?", (user_id,))
        row = await cur.fetchone()
        return row

async def save_user(user_id: int, steam_id: str = "", username: str = "", referrer_id: int = None):
    now = datetime.now().isoformat()
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("""
            INSERT OR IGNORE INTO users (user_id, steam_id, username, joined_at, referrer_id)
            VALUES (?, ?, ?, ?, ?)
        """, (user_id, steam_id, username, now, referrer_id))
        if steam_id:
            await db.execute("UPDATE users SET steam_id = ? WHERE user_id = ?", (steam_id, user_id))
        await db.commit()

async def is_premium(user_id: int, steam_id: str = "") -> bool:
    if DEBUG_MODE and steam_id == DEBUG_STEAM_ID:
        return True
    user = await get_user(user_id)
    if not user:
        return False
    until_str = user["sub_until"]
    if not until_str:
        return False
    try:
        until = datetime.fromisoformat(until_str)
        return datetime.now() < until
    except:
        return False

async def get_sub_tier(user_id: int) -> str:
    user = await get_user(user_id)
    if not user:
        return "free"
    return user["sub_tier"] or "free"

async def can_search(user_id: int, steam_id: str = "") -> bool:
    if await is_premium(user_id, steam_id):
        return True
    user = await get_user(user_id)
    if not user:
        return True
    today = datetime.now().strftime("%Y-%m-%d")
    last = user["last_search_date"] or ""
    if last != today:
        async with aiosqlite.connect(DB_PATH) as db:
            await db.execute("UPDATE users SET searches_today = 0, last_search_date = ? WHERE user_id = ?", (today, user_id))
            await db.commit()
        return True
    return (user["searches_today"] or 0) < 3

async def increment_search(user_id: int):
    today = datetime.now().strftime("%Y-%m-%d")
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("""
            INSERT INTO users (user_id, searches_today, last_search_date)
            VALUES (?, 1, ?)
            ON CONFLICT(user_id) DO UPDATE SET
                searches_today = CASE WHEN last_search_date = ? THEN searches_today + 1 ELSE 1 END,
                last_search_date = ?
        """, (user_id, today, today, today))
        await db.commit()

async def get_searches_left(user_id: int) -> int:
    user = await get_user(user_id)
    if not user:
        return 3
    today = datetime.now().strftime("%Y-%m-%d")
    if user["last_search_date"] != today:
        return 3
    return max(0, 3 - (user["searches_today"] or 0))

def get_float_tier(wear: str, float_val: float) -> float:
    rng = FLOAT_RANGES.get(wear)
    if not rng:
        return 0.0
    lo, hi = rng
    pos = (float_val - lo) / (hi - lo) if hi > lo else 0
    if pos <= 0.05:
        return 0.40
    elif pos <= 0.10:
        return 0.25
    elif pos <= 0.20:
        return 0.10
    return 0.0

def check_blue_gem(paint_seed: int) -> str:
    if paint_seed in BLUE_GEM_TIER1:
        return "Tier 1"
    elif paint_seed in BLUE_GEM_TIER2:
        return "Tier 2"
    elif paint_seed in BLUE_GEM_TIER3:
        return "Tier 3"
    return ""

def check_rare_sticker(sticker_name: str) -> dict:
    low = sticker_name.lower()
    for key, val in RARE_STICKERS.items():
        if key in low:
            return val
    return {}

def calc_sticker_value(stickers: list) -> tuple:
    total = 0
    rare_tags = []
    for s in stickers:
        name = s.get("name", "")
        wear = s.get("wear", 0) or 0
        wear_mult = max(0.1, 1.0 - wear)
        rare = check_rare_sticker(name)
        if rare:
            val = rare["min_value"] * wear_mult
            total += val
            rare_tags.append(rare["tag"])
    overpay = total * 0.40
    return total, overpay, rare_tags

def build_buy_link(platform: str, item: dict) -> str:
    name = item.get("market_hash_name", item.get("name", ""))
    if platform == "steam":
        return f"https://steamcommunity.com/market/listings/730/{quote(name)}"
    elif platform == "csfloat":

        if item_id:
            return f"https://csfloat.com/item/{item_id}"
        return f"https://csfloat.com/search?q={quote(name)}"
    elif platform == "skinport":
        asset_id = item.get("asset_id", item.get("id", ""))
        slug = name.replace("|", "").replace("(", "").replace(")", "").strip().replace(" ", "-").lower()
        if asset_id:
            return f"https://skinport.com/item/{slug}/{asset_id}"
        return f"https://skinport.com/items/{slug}"
    return "#"

async def fetch_steam(session: aiohttp.ClientSession, query: str, count: int = 30) -> list:
    url = "https://steamcommunity.com/market/search/render/"
    params = {
        "query": query,
        "appid": 730,
        "norender": 1,
        "count": count,
    }
    headers = {"User-Agent": "Mozilla/5.0"}
    try:
        async with session.get(url, params=params, headers=headers, timeout=15) as r:
            if r.status != 200:
                logger.warning(f"Steam search status {r.status}")
                return []
            data = await r.json()
            results = []
            for item in data.get("results", [])[:count]:
                name = item.get("hash_name", item.get("name", ""))
                price_raw = item.get("sell_price_text", "0")
                try:
                    price = float(item.get("sell_price", 0)) / 100.0
                except:
                    price = 0
                if price <= 0:
                    continue
                volume = item.get("volume", "0")
                try:
                    vol_num = int(volume.replace(",", "")) if isinstance(volume, str) else int(volume)
                except:
                    vol_num = 0
                results.append({
                    "platform": "steam",
                    "name": name,
                    "market_hash_name": name,
                    "price": price,
                    "currency": "RUB",
                    "volume": vol_num,
                    "id": "",
                })
            return results
    except Exception as e:
        logger.warning(f"Steam fetch error: {e}")
        return []

async def fetch_steam_priceoverview(session: aiohttp.ClientSession, name: str) -> dict:
    url = "https://steamcommunity.com/market/priceoverview/"
    params = {"appid": 730, "market_hash_name": name, "currency": 5}

async def fetch_csfloat(session: aiohttp.ClientSession, query: str, count: int = 30) -> list:
    url = "https://csfloat.com/api/v1/listings"
    params = {
        "sort_by": "most_recent",
        "min_price": 1,
        "limit": count,
        "q": query,
    }
    headers = {"User-Agent": "Mozilla/5.0"}
    try:
        async with session.get(url, params=params, headers=headers, timeout=15) as r:
            if r.status != 200:
                logger.warning(f"CSFloat status {r.status}")
                return []
            data = await r.json()
            if not isinstance(data, list):
                return []
            results = []
            for item in data[:count]:
                name = item.get("item", {}).get("market_hash_name", "")
                price_usd = item.get("price", 0)
                price_rub = price_usd * _usd_rub if price_usd else 0
                if price_rub <= 0 or not name:
                    continue
                float_val = item.get("float_value", 0) or 0
                paint_seed = item.get("paint_seed", 0) or 0
                stickers_raw = item.get("stickers", [])
                stickers_list = []
                for s in stickers_raw:
                    stickers_list.append({
                        "name": s.get("name", ""),
                        "wear": s.get("wear", 0) or 0,
                    })
                results.append({
                    "platform": "csfloat",
                    "name": name,
                    "market_hash_name": name,
                    "price": round(price_rub, 2),
                    "currency": "RUB",
                    "id": item.get("id", ""),
                    "float_value": float_val,
                    "paint_seed": paint_seed,
                    "stickers": stickers_list,
                    "volume": 0,
                })
            return results
    except Exception as e:
        logger.warning(f"CSFloat fetch error: {e}")
        return []

# ARBITRAGE SCANNER
# ═══════════════════════════════════════════

async def scan_market(query: str = "", user_id: int = 0, steam_id: str = "") -> dict:
    async with _scan_sem:
        await update_rates()
        results = {"steam": [], "csfloat": [], "skinport": [], "deals": [], "status": {}}

        async with aiohttp.ClientSession() as session:
            tasks = [
                ("steam", fetch_steam(session, query)),
                ("csfloat", fetch_csfloat(session, query)),
                ("skinport", fetch_skinport(session, query)),
            ]
            done = await asyncio.gather(*[t[1] for t in tasks], return_exceptions=True)

        for i, (platform_name, _) in enumerate(tasks):
            res = done[i]
            if isinstance(res, Exception):
                results["status"][platform_name] = {"ok": False, "count": 0}
                logger.warning(f"{platform_name} failed: {res}")
            elif isinstance(res, list):
                results[platform_name] = res
                results["status"][platform_name] = {"ok": True, "count": len(res)}
            else:
                results["status"][platform_name] = {"ok": False, "count": 0}

        # Build price map by item name
        price_map = {}
        for platform in ["steam", "csfloat", "skinport"]:
            for item in results[platform]:
                name = item["name"]
                if name not in price_map:
                    price_map[name] = {}
                price_map[name][platform] = item

        # Find arbitrage deals
        deals = []
        for name, platforms in price_map.items():
            buy_platform = None
            buy_price = float("inf")
            sell_platform = None
            sell_price = 0

            for p in ["csfloat", "skinport", "steam"]:
                if p in platforms:
                    p_price = platforms[p]["price"]
                    if p_price < buy_price:
                        buy_price = p_price
                        buy_platform = p

            for p in ["steam", "csfloat", "skinport"]:
                if p in platforms and p != buy_platform:
                    p_price = platforms[p]["price"]
                    if p_price > sell_price:
                        sell_price = p_price
                        sell_platform = p

            if not buy_platform or not sell_platform or buy_price >= sell_price:
                continue

            commission_map = {
                "steam": STEAM_COMMISSION,
                "csfloat": CSFLOAT_COMMISSION,
                "skinport": SKINPORT_COMMISSION,
            }
            commission = commission_map.get(sell_platform, 0.15)
            net_sell = sell_price * (1 - commission)
            profit = net_sell - buy_price
            roi = profit / buy_price if buy_price > 0 else 0

            threshold = get_roi_threshold(buy_price)
            if roi < threshold:
                continue

            buy_item = platforms[buy_platform]

            # Hidden value: stickers
            sticker_total = 0
            sticker_overpay = 0
            rare_tags = []
            if buy_platform == "csfloat" and buy_item.get("stickers"):
                sticker_total, sticker_overpay, rare_tags = calc_sticker_value(buy_item["stickers"])

            # Hidden value: float premium
            float_premium = 0
            float_val = buy_item.get("float_value", 0)
            wear = ""
            for w in FLOAT_RANGES:
                if w in name.lower():
                    wear = w
                    break
            if wear and float_val:
                float_premium = get_float_tier(wear, float_val) * buy_price

            # Hidden value: Blue Gem
            blue_gem = ""
            blue_gem_mult = 0
            paint_seed = buy_item.get("paint_seed", 0)
            if paint_seed:
                bg = check_blue_gem(paint_seed)
                if bg:
                    blue_gem = bg
                    if "Tier 1" in bg:
                        blue_gem_mult = 9.0
                    elif "Tier 2" in bg:
                        blue_gem_mult = 4.0
                    elif "Tier 3" in bg:
                        blue_gem_mult = 1.5

            blue_gem_value = buy_price * blue_gem_mult
            real_value = buy_price + sticker_overpay + float_premium + blue_gem_value
            hidden_value = real_value - buy_price
            hidden_pct = hidden_value / buy_price if buy_price else 0

            volume = buy_item.get("volume", 0)
            liq = liquidity_indicator(volume)

            deal = {
                "name": name,
                "buy_platform": buy_platform,
                "buy_price": round(buy_price, 2),
                "sell_platform": sell_platform,
                "sell_price": round(sell_price, 2),
                "net_sell": round(net_sell, 2),
                "profit": round(profit, 2),
                "roi": round(roi * 100, 1),
                "threshold": round(threshold * 100, 1),
                "volume": volume,
                "liquidity": liq,
                "float_value": float_val,
                "float_premium": round(float_premium, 2),
                "stickers_total": round(sticker_total, 2),
                "rare_tags": rare_tags,
                "blue_gem": blue_gem,
                "paint_seed": paint_seed,
                "hidden_value": round(hidden_value, 2),
                "hidden_pct": round(hidden_pct * 100, 1),
                "real_value": round(real_value, 2),
                "buy_item": buy_item,
                "buy_link": build_buy_link(buy_platform, buy_item),
            }
            deals.append(deal)

        deals.sort(key=lambda d: d["roi"], reverse=True)
        results["deals"] = deals
        return results

def format_deal_card(deal: dict, index: int = 1, full: bool = True) -> str:

async def send_deal_of_day():
    today = datetime.now().strftime("%Y-%m-%d")
    async with aiosqlite.connect(DB_PATH) as db:
        cur = await db.execute("SELECT * FROM deal_of_day WHERE date = ?", (today,))
        existing = await cur.fetchone()
        if existing:
            return

    await update_rates()
    try:
        result = await scan_market("")
    except:
        return

    deals = result.get("deals", [])
    if not deals:
        # Fallback: send market status
        async with aiosqlite.connect(DB_PATH) as db:
            async with db.execute("SELECT user_id FROM users") as cur:
                users = await cur.fetchall()

# PORTFOLIO MONITOR
# ═══════════════════════════════════════════

async def monitor_portfolio():
    while True:
        await asyncio.sleep(PORTFOLIO_CHECK_INTERVAL)
        try:
            async with aiosqlite.connect(DB_PATH) as db:
                db.row_factory = aiosqlite.Row
                cur = await db.execute("SELECT * FROM portfolio WHERE status = 'active'")
                positions = await cur.fetchall()

            if not positions:
                continue

            async with aiohttp.ClientSession() as session:
                for pos in positions:
                    name = pos["item_name"]
                    buy_price = pos["buy_price"]
                    try:
                        po = await fetch_steam_priceoverview(session, name)
                        current_price = po.get("median_price", 0)
                    except:
                        continue

                    if current_price <= 0:
                        continue

                    pnl_pct = (current_price - buy_price) / buy_price * 100 if buy_price else 0

                    if pnl_pct >= 15:
                        await bot.send_message(pos["user_id"],
                            f"\U0001F4C8 \u041F\u043E\u0440\u0430 \u043F\u0440\u043E\u0434\u0430\u0432\u0430\u0442\u044C!\n\n{name}\n\u041A\u0443\u043F\u043B\u0435\u043D\u043E: {buy_price:.0f}\u20BD\n\u0422\u0435\u043A\u0443\u0449\u0430\u044F: {current_price:.0f}\u20BD\n\u041F\u0440\u0438\u0431\u044B\u043B\u044C: +{current_price - buy_price:.0f}\u20BD ({pnl_pct:.1f}%)")
                    elif pnl_pct <= -10:
                        await bot.send_message(pos["user_id"],
                            f"\U0001F6D1 \u0421\u0442\u043E\u043F-\u043B\u043E\u0441\u0441!\n\n{name}\n\u041A\u0443\u043F\u043B\u0435\u043D\u043E: {buy_price:.0f}\u20BD\n\u0422\u0435\u043A\u0443\u0449\u0430\u044F: {current_price:.0f}\u20BD\n\u0423\u0431\u044B\u0442\u043E\u043A: {current_price - buy_price:.0f}\u20BD ({pnl_pct:.1f}%)\n\n\u0420\u0435\u043A\u043E\u043C\u0435\u043D\u0434\u0443\u0435\u043C \u043F\u0440\u043E\u0434\u0430\u0442\u044C.")

                    # Trade ban ending
                    if pos["trade_ban_until"]:
                        try:
                            tb = datetime.fromisoformat(pos["trade_ban_until"])
                            days_left = (tb - datetime.now()).days
                            if days_left == 1:
                                await bot.send_message(pos["user_id"],
                                    f"\u23F0 \u0422\u0440\u0435\u0439\u0434-\u0431\u0430\u043D \u0437\u0430\u043A\u0430\u043D\u0447\u0438\u0432\u0430\u0435\u0442\u0441\u044F \u0437\u0430\u0432\u0442\u0440\u0430.\n\n{name}\n\u0422\u0435\u043A\u0443\u0449\u0430\u044F \u0446\u0435\u043D\u0430: {current_price:.0f}\u20BD\n\u0420\u0435\u043A\u043E\u043C\u0435\u043D\u0434\u0430\u0446\u0438\u044F: \u0432\u044B\u0441\u0442\u0430\u0432\u0438\u0442\u044C \u043D\u0430 \u043F\u0440\u043E\u0434\u0430\u0436\u0443.")
                            elif days_left <= 0:
                                await bot.send_message(pos["user_id"],
                                    f"\u2705 \u0422\u0440\u0435\u0439\u0434-\u0431\u0430\u043D \u0438\u0441\u0442\u0451\u043A!\n\n{name}\n\u0422\u0435\u043A\u0443\u0449\u0430\u044F \u0446\u0435\u043D\u0430: {current_price:.0f}\u20BD\n\u041C\u043E\u0436\u043D\u043E \u043F\u0440\u043E\u0434\u0430\u0432\u0430\u0442\u044C!")
                                await db.execute("UPDATE portfolio SET status = 'tradable' WHERE id = ?", (pos["id"],))
                                await db.commit()
                        except:
                            pass
        except Exception as e:
            logger.error(f"Portfolio monitor error: {e}")

async def scheduled_tasks():
    while True:
        await asyncio.sleep(60)
        now = datetime.now()
        # Deal of day at 12:00
        if now.hour == 12 and now.minute < 2:
            await send_deal_of_day()
            await asyncio.sleep(120)

    global bot
    if BOT_TOKEN == "TOKEN_HERE":
