_currency_cache = {"usd": 0, "eur": 0, "ts": 0}

async def fetch_currency_rates():
    now = time.time()
    if _currency_cache["ts"] and now - _currency_cache["ts"] < 3600:
        return _currency_cache["usd"], _currency_cache["eur"]
    try:
        async with aiohttp.ClientSession() as session:
            async with session.get("https://open.er-api.com/v6/latest/USD", timeout=10) as resp:
                data = await resp.json()
                usd = data.get("rates", {}).get("RUB", USD_RUB_FALLBACK)
            async with session.get("https://open.er-api.com/v6/latest/EUR", timeout=10) as resp:
                data = await resp.json()
                eur = data.get("rates", {}).get("RUB", EUR_RUB_FALLBACK)
        _currency_cache.update({"usd": usd, "eur": eur, "ts": now})
        logger.info(f"Currency updated: USD={usd}, EUR={eur}")
        return usd, eur
    except Exception as e:
        logger.warning(f"Currency fetch failed: {e}")
        usd = _currency_cache["usd"] or USD_RUB_FALLBACK
        eur = _currency_cache["eur"] or EUR_RUB_FALLBACK
        return usd, eur

def evaluate_stickers(stickers_data, usd_rub):
    if not stickers_data or not isinstance(stickers_data, list):
        return {"count": 0, "total_value_rub": 0, "rare_flags": [], "has_rare": False, "detail": ""}
    
    total_value = 0
    rare_flags = []
    detail_parts = []
    
    for st in stickers_data:
        name = st.get("name", "").lower()
        wear = st.get("wear", 0)
        
        is_rare = False
        for rare_key, rare_info in RARE_STICKERS.items():
            if rare_key in name:
                is_rare = True
                rare_flags.append(rare_info["tag"])
                wear_factor = max(0.1, 1.0 - wear) if wear else 1.0
                st_value = rare_info["min_value"] * wear_factor
                total_value += st_value
                detail_parts.append(f"{rare_info['tag']} (~{int(st_value)}rub)")
                break
        
        if not is_rare:
            wear_factor = max(0.1, 1.0 - wear) if wear else 1.0
            st_value = 100 * wear_factor
            total_value += st_value
    
    overpay_value = total_value * 0.4
    
    return {
        "count": len(stickers_data),
        "total_value_rub": int(overpay_value),
        "raw_value_rub": int(total_value),
        "rare_flags": rare_flags,
        "has_rare": len(rare_flags) > 0,
        "detail": " | ".join(detail_parts) if detail_parts else ""
    }

# FLOAT-ПРЕМИУМ
# ============================================================

def calculate_float_premium(float_value, wear_name):
    if not float_value or not wear_name:
        return 0
    
    wear_lower, wear_upper = WEAR_RANGES.get(wear_name, (0, 1))
    wear_range = wear_upper - wear_lower
    if wear_range == 0:
        return 0
    
    position = (float_value - wear_lower) / wear_range
    
    if position <= 0.05:
        return 0.40
    elif position <= 0.10:
        return 0.25
    elif position <= 0.20:
        return 0.10
    return 0

def detect_blue_gem(paint_seed, item_name):
    if not paint_seed:
        return {"is_blue_gem": False, "tier": 0, "multiplier": 1.0, "tag": ""}
    
    name_lower = (item_name or "").lower()
    seed = int(paint_seed)
    
    if "case hardened" not in name_lower:
        return {"is_blue_gem": False, "tier": 0, "multiplier": 1.0, "tag": ""}
    
    if seed in BLUE_GEM_TIER1:
        return {"is_blue_gem": True, "tier": 1, "multiplier": BLUE_GEM_TIERS["tier1"]["multiplier"],
                "tag": BLUE_GEM_TIERS["tier1"]["tag"]}
    
    if seed in BLUE_GEM_TIER2:
        return {"is_blue_gem": True, "tier": 2, "multiplier": BLUE_GEM_TIERS["tier2"]["multiplier"],
                "tag": BLUE_GEM_TIERS["tier2"]["tag"]}
    
    if seed in BLUE_GEM_TIER3:
        return {"is_blue_gem": True, "tier": 3, "multiplier": BLUE_GEM_TIERS["tier3"]["multiplier"],
                "tag": BLUE_GEM_TIERS["tier3"]["tag"]}
    
    return {"is_blue_gem": False, "tier": 0, "multiplier": 1.0, "tag": ""}

# РАСЧЁТ РЕАЛЬНОЙ СТОИМОСТИ
# ============================================================

def calculate_real_value(base_price_rub, float_value, wear_name, stickers_data, paint_seed, item_name, usd_rub):
    sticker_info = evaluate_stickers(stickers_data, usd_rub)
    float_premium_pct = calculate_float_premium(float_value, wear_name)
    blue_gem_info = detect_blue_gem(paint_seed, item_name)
    
    float_bonus = base_price_rub * float_premium_pct
    
    blue_gem_bonus = 0
    if blue_gem_info["is_blue_gem"]:
        blue_gem_bonus = base_price_rub * (blue_gem_info["multiplier"] - 1.0)
    
    sticker_bonus = sticker_info["total_value_rub"]
    
    real_value = base_price_rub + float_bonus + blue_gem_bonus + sticker_bonus
    
    hidden_value = real_value - base_price_rub
    hidden_pct = (hidden_value / base_price_rub * 100) if base_price_rub > 0 else 0
    
    return {
        "real_value": int(real_value),
        "base_price": int(base_price_rub),
        "sticker_info": sticker_info,
        "float_premium_pct": float_premium_pct,
        "float_bonus": int(float_bonus),
        "blue_gem_info": blue_gem_info,
        "blue_gem_bonus": int(blue_gem_bonus),
        "sticker_bonus": sticker_bonus,
        "hidden_value": int(hidden_value),
        "hidden_pct": round(hidden_pct, 1),
        "has_hidden_value": hidden_value > base_price_rub * 0.15
    }

def parse_price_text(text):
    if not text:
        return 0
    cleaned = re.sub(r"[^0-9.,]", "", str(text).replace(",", "."))
    try:
        return float(cleaned)
    except:
        return 0

async def fetch_steam_search(game, start, count, session):
    app_id = 730 if game == "cs2" else 570
    url = "https://steamcommunity.com/market/search/render/"
    params = {"appid": app_id, "norender": 1, "count": count, "start": start}
    try:
        headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"}
        async with session.get(url, params=params, headers=headers, timeout=15) as resp:
            if resp.status != 200:
                return []
            data = await resp.json()
            results = data.get("results", [])
            items = []
            for r in results:
                name = r.get("hash_name", r.get("name", ""))
                price = r.get("sell_price_text", "0")
                price_num = parse_price_text(price)
                listings = r.get("sell_listings", 0)
                items.append({"name": name, "price": price_num, "listings": listings, "source": "steam"})
            return items
    except Exception as e:
        logger.warning(f"Steam search error: {e}")
        return []

async def fetch_steam_priceoverview(market_hash, session):
    url = "https://steamcommunity.com/market/priceoverview/"
    params = {"appid": 730, "market_hash_name": market_hash}
    try:
        headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"}
        async with session.get(url, params=params, headers=headers, timeout=10) as resp:
            if resp.status != 200:
                return None
            data = await resp.json()
            median = parse_price_text(data.get("median_price", "0"))
            volume = 0
            vol_str = data.get("volume", "0")
            try:
                volume = int(vol_str)
            except:
                volume = 0
            return {"median": median, "volume": volume}
    except:
        return None

async def fetch_csfloat_listings(category, min_price, max_price, session):
    url = "https://csfloat.com/api/v1/listings"
    params = {"sort_by": "most_recent", "min_price": min_price, "max_price": max_price,
              "category": category, "limit": 50}
    try:
        headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"}
        async with session.get(url, params=params, headers=headers, timeout=15) as resp:
            if resp.status != 200:
                return []
            data = await resp.json()
            items = []
            for item in data:
                item_data = item.get("item", {})
                name = item_data.get("market_hash_name", "")
                price_usd = item.get("price", 0) / 100
                float_value = item_data.get("float_value", 0)
                paint_seed = item_data.get("paint_seed", 0)
                stickers = item_data.get("stickers", [])
                wear_name = item_data.get("wear_name", "")
                asset_id = item.get("id", "")
                items.append({
                    "name": name, "price_usd": price_usd,
                    "price_rub": price_usd * get_usd_rub(),
                    "float_value": float_value, "paint_seed": paint_seed,
                    "stickers": stickers, "wear_name": wear_name,
                    "asset_id": asset_id, "source": "csfloat",
                    "url": f"https://csfloat.com/item/{asset_id}" if asset_id else "https://csfloat.com"
                })
            return items
    except Exception as e:
        logger.warning(f"CSFloat error: {e}")
        return []

_scan_cache = {}

async def scan_market(segment, usd_rub):
    async with SCAN_SEMAPHORE:
        cache_key = f"{segment}_{int(usd_rub)}"
        now = time.time()
        if cache_key in _scan_cache and now - _scan_cache[cache_key]["ts"] < SCAN_CACHE_TTL:
            return _scan_cache[cache_key]["data"]
        
        seg_ranges = {"low": (0, 500), "mid": (500, 2000), "high": (2000, 100000)}
        min_rub, max_rub = seg_ranges.get(segment, (500, 2000))
        min_usd = min_rub / usd_rub
        max_usd = max_rub / usd_rub
        
        statuses = {}
        all_deals = []
        
        async with aiohttp.ClientSession() as session:
            steam_task = fetch_steam_search("cs2", 0, 100, session)
            csfloat_task = fetch_csfloat_listings(0, min_usd * 100, max_usd * 100, session)
            skinport_task = fetch_skinport_items(session)
            
            steam_items, csfloat_items, skinport_items = await asyncio.gather(
                steam_task, csfloat_task, skinport_task, return_exceptions=True
            )
            
            if isinstance(steam_items, Exception) or not steam_items:
                statuses["steam"] = {"ok": False, "count": 0}
                steam_items = []
            else:
                statuses["steam"] = {"ok": True, "count": len(steam_items)}
            
            if isinstance(csfloat_items, Exception) or not csfloat_items:
                statuses["csfloat"] = {"ok": False, "count": 0}
                csfloat_items = []
            else:
                statuses["csfloat"] = {"ok": True, "count": len(csfloat_items)}
            
            if isinstance(skinport_items, Exception) or not skinport_items:
                statuses["skinport"] = {"ok": False, "count": 0}
                skinport_items = []
            else:
                statuses["skinport"] = {"ok": True, "count": len(skinport_items)}
            
            steam_prices = {}
            for item in steam_items:
                if min_rub <= item["price"] <= max_rub:
                    steam_prices[item["name"]] = item["price"]
            
            steam_overview = {}
            names_to_check = list(steam_prices.keys())[:30]
            for name in names_to_check:
                ov = await fetch_steam_priceoverview(name, session)
                if ov:
                    steam_overview[name] = ov
                await asyncio.sleep(3)
            
            # CSFloat deals with sniper analysis
            for cf_item in csfloat_items:
                name = cf_item["name"]
                buy_price = cf_item["price_rub"]
                if buy_price < min_rub or buy_price > max_rub:
                    continue
                
                real_val = calculate_real_value(
                    buy_price, cf_item.get("float_value"),
                    cf_item.get("wear_name"), cf_item.get("stickers"),
                    cf_item.get("paint_seed"), name, usd_rub
                )
                
                sell_steam_median = steam_overview.get(name, {}).get("median", 0)
                sell_price = max(sell_steam_median, real_val["real_value"]) if sell_steam_median > 0 else real_val["real_value"]
                
                if real_val["has_hidden_value"]:
                    sell_price = real_val["real_value"]
                
                if sell_price <= 0 or buy_price <= 0:
                    continue
                
                commission = sell_price * STEAM_COMMISSION
                net = sell_price - commission
                profit = net - buy_price
                roi = profit / buy_price if buy_price > 0 else 0
                volume = steam_overview.get(name, {}).get("volume", 0)
                
                if roi >= MIN_ROI_THRESHOLD or real_val["has_hidden_value"]:
                    all_deals.append({
                        "name": name, "buy_price": buy_price, "sell_price": sell_price,
                        "base_sell": sell_steam_median, "commission": commission,
                        "net": net, "profit": profit, "roi": roi,
                        "strategy": "csfloat_to_steam", "strategy_icon": "rocket",
                        "source": "csfloat",
                        "url": cf_item.get("url", "https://csfloat.com"),
                        "float_value": cf_item.get("float_value"),
                        "wear_name": cf_item.get("wear_name"),
                        "paint_seed": cf_item.get("paint_seed"),
                        "stickers": cf_item.get("stickers"),
                        "volume": volume, "real_value": real_val,
                    })
            
            # Skinport deals
            for sp_item in skinport_items:
                name = sp_item["name"]
                buy_price = sp_item["price"]
                if buy_price < min_rub or buy_price > max_rub:
                    continue
                sell_steam_median = steam_overview.get(name, {}).get("median", 0)
                if sell_steam_median <= 0:
                    continue
                commission = sell_steam_median * STEAM_COMMISSION
                net = sell_steam_median - commission
                profit = net - buy_price
                roi = profit / buy_price if buy_price > 0 else 0
                volume = steam_overview.get(name, {}).get("volume", 0)
                if roi >= MIN_ROI_THRESHOLD:
                    all_deals.append({
                        "name": name, "buy_price": buy_price, "sell_price": sell_steam_median,
                        "base_sell": sell_steam_median, "commission": commission,
                        "net": net, "profit": profit, "roi": roi,
                        "strategy": "skinport_to_steam", "strategy_icon": "package",
                        "source": "skinport",
                        "url": sp_item.get("url", "https://skinport.com"),
                        "float_value": None, "wear_name": "", "paint_seed": None,
                        "stickers": [], "volume": volume, "real_value": None,
                    })
            
            # Steam flip
            for name, buy_price in steam_prices.items():
                if buy_price < min_rub or buy_price > max_rub:
                    continue
                ov = steam_overview.get(name, {})
                median = ov.get("median", 0)
                volume = ov.get("volume", 0)
                if median <= 0 or buy_price <= 0:
                    continue
                commission = median * STEAM_COMMISSION
                net = median - commission
                profit = net - buy_price
                roi = profit / buy_price if buy_price > 0 else 0
                if roi >= MIN_ROI_THRESHOLD:
                    all_deals.append({
                        "name": name, "buy_price": buy_price, "sell_price": median,
                        "base_sell": median, "commission": commission,
                        "net": net, "profit": profit, "roi": roi,
                        "strategy": "steam_flip", "strategy_icon": "rotate",
                        "source": "steam",
                        "url": f"https://steamcommunity.com/market/listings/730/{urllib.parse.quote(name)}",
                        "float_value": None, "wear_name": "", "paint_seed": None,
                        "stickers": [], "volume": volume, "real_value": None,
                    })
        
        all_deals.sort(key=lambda d: (
            -1 if d.get("real_value") and d["real_value"]["has_hidden_value"] else 0,
            d["roi"]
        ), reverse=True)
        
        result = {"deals": all_deals[:20], "statuses": statuses}
        _scan_cache[cache_key] = {"data": result, "ts": now}
        return result

def format_deal(deal, index, usd_rub):
    lines = []
    lines.append(f"\n**{index}. {deal['name']}**\n")
    lines.append(f"Strategy: {deal['strategy_icon']} {strategy_name(deal['strategy'])}")
    lines.append(f"Buy ({deal['source']}): {fmt_price(deal['buy_price'])}")
    lines.append(f"Sell (Steam): {fmt_price(deal['sell_price'])}")
    lines.append(f"Commission (15%): -{fmt_price(deal['commission'])}")
    lines.append(f"Net: {fmt_price(deal['net'])}")
    profit_emoji = "+" if deal['profit'] > 0 else "-"
    lines.append(f"Profit: {profit_emoji}{fmt_price(deal['profit'])} ({fmt_pct(deal['roi'] * 100)})")
    lines.append(f"Liquidity: {get_liquidity_text(deal.get('volume', 0))}")
    
    if deal.get("float_value") is not None:
        float_str = f"{deal['float_value']:.4f}"
        wear = deal.get("wear_name", "")
        lines.append(f"Float: {float_str} ({wear})")
    
    rv = deal.get("real_value")
    if rv and rv["sticker_info"]["count"] > 0:
        st_info = rv["sticker_info"]
        lines.append(f"Stickers: {st_info['count']}")
        if st_info["has_rare"]:
            lines.append(f"RARE: {', '.join(st_info['rare_flags'])}")
        if st_info["detail"]:
            lines.append(f"  {st_info['detail']}")
    
    if rv and rv["blue_gem_info"]["is_blue_gem"]:
        bg = rv["blue_gem_info"]
        lines.append(f"{bg['tag']} (seed: {deal.get('paint_seed', '?')})")
    
    if rv and rv["float_premium_pct"] > 0:
        lines.append(f"Float premium: +{fmt_pct(rv['float_premium_pct'] * 100)}")
    
    if rv and rv["has_hidden_value"]:
        lines.append(f"\nHIDDEN VALUE: +{fmt_price(rv['hidden_value'])} ({fmt_pct(rv['hidden_pct'])})")
        lines.append(f"Real value: {fmt_price(rv['real_value'])}")
    
    lines.append(f"\nTrade ban: 7 days. Price may change.")
    
    lines.append(f"\nDeal check:")
    lines.append(f"ROI: {fmt_pct(deal['roi'] * 100)} (threshold 15%)")
    lines.append(f"Commission: Steam 15% included")
    liq = get_liquidity_emoji(deal.get("volume", 0))
    lines.append(f"Liquidity: {liq}")
    if deal.get("float_value") is not None:
        lines.append(f"Float: {deal['float_value']:.4f}")
    if rv and rv["sticker_info"]["has_rare"]:
        lines.append(f"Rare stickers: {', '.join(rv['sticker_info']['rare_flags'])}")
    if rv and rv["blue_gem_info"]["is_blue_gem"]:
        lines.append(f"Blue Gem: {rv['blue_gem_info']['tag']}")
    lines.append(f"Trade ban: 7 days")
    status = "PROFITABLE" if deal['roi'] >= MIN_ROI_THRESHOLD else "CAUTION"
    lines.append(f"\nStatus: {status}")
    
    return "\n".join(lines)

async def get_user(user_id):
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        async with db.execute("SELECT * FROM users WHERE user_id = ?", (user_id,)) as cur:
            row = await cur.fetchone()
            if not row:
                now = time.time()
                await db.execute(
                    "INSERT INTO users (user_id, first_seen, created_at) VALUES (?, ?, ?)",
                    (user_id, now, now))
                await db.commit()
                return {"user_id": user_id, "steam_id": None, "game": "cs2",
                        "segment": "mid", "sub_tier": "freemium", "sub_until": 0,
                        "searches_today": 0, "searches_reset_day": 0, "first_seen": now,
                        "referrer": "", "trial_used": 0, "last_remind_day": 0, "created_at": now}
            return dict(row)

async def update_user(user_id, **kwargs):
    cols = ", ".join(f"{k} = ?" for k in kwargs)
    vals = list(kwargs.values()) + [user_id]
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute(f"UPDATE users SET {cols} WHERE user_id = ?", vals)
        await db.commit()

async def is_subscribed(user_id):
    user = await get_user(user_id)
    now = time.time()
    if user["sub_tier"] in ("start", "pro", "max") and user["sub_until"] > now:
        return True, user["sub_tier"]
    return False, user.get("sub_tier", "freemium")

async def check_freemium_limit(user_id):
    user = await get_user(user_id)
    today = datetime.now().day
    if user["searches_reset_day"] != today:
        await update_user(user_id, searches_today=0, searches_reset_day=today)
        user["searches_today"] = 0
    if user["sub_tier"] in ("start", "pro", "max") and user["sub_until"] > time.time():
        return True, -1
    if user["searches_today"] < FREEMIUM_DAILY_LIMIT:
        return True, FREEMIUM_DAILY_LIMIT - user["searches_today"] - 1
    return False, 0

async def increment_searches(user_id):
    user = await get_user(user_id)
    today = datetime.now().day
    if user["searches_reset_day"] != today:
        await update_user(user_id, searches_today=1, searches_reset_day=today)
    else:
        await update_user(user_id, searches_today=user["searches_today"] + 1)

async def save_deal_to_history(user_id, deal):

        rv = deal.get("real_value")

             rv["sticker_info"]["count"] if rv else 0,

             rv["blue_gem_info"]["tag"] if rv and rv["blue_gem_info"]["is_blue_gem"] else "",
             time.time()))

async def send_deal_of_the_day():
    usd_rub = get_usd_rub()
    best_deal = None
    for seg in ["low", "mid", "high"]:
        result = await scan_market(seg, usd_rub)
        deals = result.get("deals", [])
        if deals:
            hidden = [d for d in deals if d.get("real_value") and d["real_value"]["has_hidden_value"]]
            if hidden:
                best_deal = hidden[0]
                break
            if not best_deal or deals[0]["roi"] > best_deal["roi"]:
                best_deal = deals[0]
    
    if not best_deal:
        logger.info("Deal of the day: market empty")
        return
    
    is_below = best_deal["roi"] < MIN_ROI_THRESHOLD
    header = "DEAL OF THE DAY (free)\n\n" if not is_below else              "Best deal today (ROI below 15% threshold)\n\n"
    
    text = header + format_deal(best_deal, 1, usd_rub)
    text += "\n\nWant more? Subscribe -> /subscribe"
    
    kb = InlineKeyboardBuilder()
    kb.button(text=f"Buy on {best_deal['source'].capitalize()}", url=best_deal["url"])
    kb.button(text="Subscribe", callback_data="subscribe")
    kb.adjust(1, 1)
    
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        async with db.execute("SELECT user_id FROM users") as cur:
            users = await cur.fetchall()
    
    sent = 0
    for u in users:
        try:
            await bot.send_message(u["user_id"], text, reply_markup=kb.as_markup())
            sent += 1
            await asyncio.sleep(0.05)
        except:
            pass
    logger.info(f"Deal of the day sent to {sent} users")

async def check_stoploss():
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        async with db.execute("SELECT * FROM favorites") as cur:
            favs = await cur.fetchall()
    if not favs:
        return
    async with aiohttp.ClientSession() as session:
        for fav in favs:
            ov = await fetch_steam_priceoverview(fav["market_hash"], session)
            if not ov:
                continue
            current = ov["median"]
            added = fav["added_price"]
            if added <= 0:
                continue
            drop = (current - added) / added
            if drop <= -STOPLoss_THRESHOLD:
                async with aiosqlite.connect(DB_PATH) as db:
                    async with db.execute(
                        "SELECT last_alert FROM stoploss_alerts WHERE user_id = ? AND market_hash = ?",
                        (fav["user_id"], fav["market_hash"])) as cur:
                        row = await cur.fetchone()
                    if row and time.time() - row[0] < 3600:
                        continue
                    await db.execute(
                        "INSERT OR REPLACE INTO stoploss_alerts (user_id, market_hash, last_alert) VALUES (?, ?, ?)",
                        (fav["user_id"], fav["market_hash"], time.time()))
                    await db.commit()
                drop_pct = abs(drop * 100)
                text = (
                    f"STOP-LOSS TRIGGERED!\n\n"
                    f"Item: {fav['market_hash']}\n"
                    f"Added at: {fmt_price(added)}\n"
                    f"Current: {fmt_price(current)}\n"
                    f"Drop: -{drop_pct:.1f}%\n\n"
                    f"Recommend selling before price drops further.\n"
                    f"Steam: https://steamcommunity.com/market/listings/730/{urllib.parse.quote(fav['market_hash'])}")
                try:
                    await bot.send_message(fav["user_id"], text)
                except:
                    pass
            await asyncio.sleep(3)

async def check_freemium_remind():
    now = time.time()
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        async with db.execute(
            "SELECT * FROM users WHERE sub_tier = 'freemium' AND first_seen > 0") as cur:
            users = await cur.fetchall()
        for user in users:
            days_active = (now - user["first_seen"]) / 86400
            if days_active >= FREEMIUM_REMIND_DAY and user["last_remind_day"] == 0:
                async with db.execute(
                    "SELECT COUNT(*) as cnt FROM deals_history WHERE user_id = ?",
                    (user["user_id"],)) as cur:
                    r = await cur.fetchone()
                text = (
                    f"You used {user['searches_today'] * 4} free searches.\n"
                    f"Bot found {r['cnt'] if r else 0} deals.\n\n"
                    f"Subscribe to not miss the next ones:\n"
                    f"Start - 99 stars/wk (unlimited)\n"
                    f"Pro - 199 stars/wk (+ alerts + stop-loss)")
                try:
                    kb = InlineKeyboardBuilder()
                    kb.button(text="Subscribe", callback_data="subscribe")
                    await bot.send_message(user["user_id"], text, reply_markup=kb.as_markup())
                    await update_user(user["user_id"], last_remind_day=1)
                except:
                    pass

async def background_tasks():
    await asyncio.sleep(10)
    while True:
        now = datetime.now()
        hour = now.hour
        if now.minute == 0:
            await fetch_currency_rates()
        if hour == DEAL_OF_THE_DAY_HOUR and now.minute < 5:
            await send_deal_of_the_day()
            await asyncio.sleep(300)
        if now.minute % 30 == 0 and now.second < 10:
            await check_stoploss()
            await asyncio.sleep(60)
        if now.minute == 15:
            await check_freemium_remind()
            await asyncio.sleep(60)
        await asyncio.sleep(30)
